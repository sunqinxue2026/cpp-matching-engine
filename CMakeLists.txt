cmake_minimum_required(VERSION 3.20)
project(cpp_matching_engine VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(order_model src/Order.cpp src/OrderBook.cpp src/OrderManager.cpp src/MatchingEngine.cpp src/TradeManager.cpp src/MarketDepth.cpp)
target_include_directories(order_model PUBLIC include)
if(MSVC)
  target_compile_options(order_model PRIVATE /W4)
else()
  target_compile_options(order_model PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(matching_engine src/main.cpp)
target_link_libraries(matching_engine PRIVATE order_model)
include(CTest)
if(BUILD_TESTING)
  add_executable(order_model_tests tests/OrderModelTests.cpp)
  target_link_libraries(order_model_tests PRIVATE order_model)
  add_test(NAME order_model_contract COMMAND order_model_tests)
  add_executable(order_book_tests tests/OrderBookTests.cpp)
  target_link_libraries(order_book_tests PRIVATE order_model)
  add_test(NAME order_book_contract COMMAND order_book_tests)
  add_executable(order_manager_tests tests/OrderManagerTests.cpp)
  target_link_libraries(order_manager_tests PRIVATE order_model)
  add_test(NAME order_manager_contract COMMAND order_manager_tests)
  add_executable(matching_engine_tests tests/MatchingEngineTests.cpp)
  target_link_libraries(matching_engine_tests PRIVATE order_model)
  add_test(NAME matching_engine_contract COMMAND matching_engine_tests)
  add_executable(trade_manager_tests tests/TradeManagerTests.cpp)
  target_link_libraries(trade_manager_tests PRIVATE order_model)
  add_test(NAME trade_manager_contract COMMAND trade_manager_tests)
  add_executable(cancel_depth_tests tests/CancelDepthTests.cpp)
  target_link_libraries(cancel_depth_tests PRIVATE order_model)
  add_test(NAME cancel_depth_contract COMMAND cancel_depth_tests)
endif()
